# === ODONTOGEST — DEPENDENCIAS BACKEND ===

# --- Core ---
fastapi[all]
# Web framework. Incluye uvicorn, starlette, cors, etc.

sqlalchemy
# ORM para modelos de base de datos relacionales

pydantic
# Validación de datos (schemas Request/Response)

tzdata
# Timezone database para zoneinfo (necesario en Windows para America/Argentina/Buenos_Aires)

pydantic-settings
# Carga de settings desde .env con tipado estricto

python-dotenv
# Carga variables de entorno desde archivo .env

# --- Base de Datos ---
psycopg2-binary
# Driver PostgreSQL (producción). En dev se usa SQLite (no requiere driver extra)

alembic
# Migraciones de base de datos. Control de versiones sobre el schema SQL.

python-jose[cryptography]

slowapi
# Rate limiting para endpoints públicos.

passlib[bcrypt]
# Hashing de contraseñas.
